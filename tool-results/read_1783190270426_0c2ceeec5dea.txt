    64→function getTimeSince(dateStr: string): string {
    65→  const now = new Date();
    66→  const date = new Date(dateStr);
    67→  const diffMs = now.getTime() - date.getTime();
    68→  const diffMin = Math.floor(diffMs / 60000);
    69→  const diffHour = Math.floor(diffMin / 60);
    70→  const diffDay = Math.floor(diffHour / 24);
    71→
    72→  if (diffMin < 1) return 'الآن';
    73→  if (diffMin < 60) return `منذ ${diffMin} دقيقة`;
    74→  if (diffHour < 24) return `منذ ${diffHour} ساعة`;
    75→  if (diffDay < 7) return `منذ ${diffDay} يوم`;
    76→  return `منذ ${Math.floor(diffDay / 7)} أسبوع`;
    77→}

    79→function formatDate(dateStr: string): string {

    82→    year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit',
    83→  });
    84→}

   125→function generateWhatsAppMessage(order: DemoOrder): string {
   126→  let templateKey = order.status;
   127→  if (order.status === 'preparing' || order.status === 'ready_delivery' || order.status === 'paid') {
   128→    templateKey = 'payment_confirmed';
   129→  }
   130→  const template = WHATSAPP_TEMPLATES[templateKey] || WHATSAPP_TEMPLATES.new_order;
   131→  return template
   132→    .replace(/{customer_name}/g, order.customerName)
   133→    .replace(/{store_name}/g, 'بوتيك النخبة')
   134→    .replace(/{order_number}/g, order.orderNumber)
   135→    .replace(/{order_total}/g, formatMRU(order.total))
   136→    .replace(/{payment_status}/g, PAYMENT_STATUS_LABELS[order.paymentStatus])
   137→    .replace(/{delivery_area}/g, order.customerArea || '')
   138→    .replace(/{store_whatsapp}/g, '22222333');
   139→}
