   330→// ============ CART STATE (client-side) ============
   331→export let cartItems: CartItem[] = [];
   332→
   333→export function addToCart(item: CartItem) {
   334→  const existing = cartItems.find(c => c.productId === item.productId && c.size === item.size && c.color === item.color);
   335→  if (existing) {
   336→    existing.quantity += item.quantity;
   337→  } else {
   338→    cartItems.push({ ...item });
   339→  }
   340→  return cartItems;
   341→}
   342→
   343→export function removeFromCart(productId: string, size: string | null, color: string | null) {
   344→  cartItems = cartItems.filter(c => !(c.productId === productId && c.size === size && c.color === color));
   345→  return cartItems;
   346→}
   347→
   348→export function updateCartQty(productId: string, size: string | null, color: string | null, qty: number) {
   349→  const item = cartItems.find(c => c.productId === productId && c.size === size && c.color === color);
   350→  if (item) item.quantity = Math.max(1, qty);
   351→  return cartItems;
   352→}
   353→
   354→export function clearCart() {
   355→  cartItems = [];
   356→  return cartItems;
   357→}
   358→
   359→export function getCartTotal(): number {
   360→  return cartItems.reduce((sum, item) => sum + item.price * item.quantity, 0);
   361→}
