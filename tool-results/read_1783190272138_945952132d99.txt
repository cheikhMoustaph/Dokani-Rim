    85→
    86→function ProgressBar({ step }: { step: number }) {
    87→  return (
    88→    <div className="w-full mb-8">
    89→      {/* Step indicators */}
    90→      <div className="flex items-center justify-between mb-3">
    91→        {STEP_LABELS.map((label, i) => {
    92→          const stepNum = i + 1;
    93→          const isActive = stepNum === step;
    94→          const isDone = stepNum < step;
    95→          return (
    96→            <div key={i} className="flex items-center flex-1 last:flex-none">
    97→              <div className="flex flex-col items-center gap-1.5">
    98→                <div
    99→                  className={`w-9 h-9 rounded-full flex items-center justify-center text-sm font-bold transition-all duration-500 ${
   100→                    isDone
   101→                      ? 'bg-[#0F7A4F] text-white shadow-md shadow-[#0F7A4F]/20'
   102→                      : isActive
   103→                        ? 'bg-[#0F7A4F] text-white shadow-lg shadow-[#0F7A4F]/30 ring-4 ring-[#0F7A4F]/10'
   104→                        : 'bg-gray-100 text-gray-400'
   105→                  }`}
   106→                >
   107→                  {isDone ? <Check className="h-4 w-4" /> : stepNum}
   108→                </div>
   109→                <span
   110→                  className={`text-[10px] sm:text-xs font-medium hidden sm:block ${
   111→                    isActive ? 'text-[#0F7A4F]' : isDone ? 'text-gray-500' : 'text-gray-400'
   112→                  }`}
   113→                >
   114→                  {label}
   115→                </span>
   116→              </div>
   117→              {i < STEP_LABELS.length - 1 && (
   118→                <div className="flex-1 mx-2 sm:mx-3">
   119→                  <div className="h-1 bg-gray-100 rounded-full overflow-hidden">
   120→                    <div
   121→                      className="h-full bg-[#0F7A4F] rounded-full transition-all duration-700 ease-out"
   122→                      style={{ width: isDone ? '100%' : '0%' }}
   123→                    />
   124→                  </div>
   125→                </div>
   126→              )}
   127→            </div>
   128→          );
   129→        })}

   131→    </div>
   132→  );
   133→}
